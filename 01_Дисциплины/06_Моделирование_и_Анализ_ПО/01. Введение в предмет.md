- [ ] #task Конспект по С++ должен включать: типы данных и сводку по ним, работу с функциями, работа с переменными, математические операции, встроенные библиотеки, как с ними работать и что они могут делать
```cpp
#include <iostream>
#include <string>
#include <windows.h>
#include <numbers>

double sphereArea(double radius) {
    double area = std::numbers::pi * 4 * pow(radius, 2);
    return area;
}
double sphereVolume(double radius) {
    double volume = std::numbers::pi * 4 / 3 * pow(radius, 3);
    return volume;
}

int main() {
    SetConsoleOutputCP(CP_UTF8);
    SetConsoleCP(CP_UTF8);

    double radius;
    std::cout << "Введите радиус: ";
    std::cin >> radius;

    double volume = sphereVolume(radius);
    double area = sphereArea(radius);

    std::cout << "Площадь: " << area << std::endl;
    std::cout << "Объём: " << volume << std::endl;

    return 0;
}
```
